# --- WebDAV / Nextcloud / ownCloud destination support -----------------------
# webdavclient3 is required for full Nextcloud / ownCloud compatibility
# (chunked uploads, lock handling, edge-case auth flows).
# transport_utils.py falls back to the built-in urllib client if it is absent,
# but webdavclient3 is strongly recommended and installed by default.
webdavclient3>=3.14.6,<4.0

# --- Secure credential storage (recommended) ---------------------------------
# Stores SFTP / FTP / SMTP passwords in the OS keyring instead of
# plaintext in config.json.  Falls back gracefully if not installed.
#   Windows  : Windows Credential Manager
#   macOS    : macOS Keychain
#   Linux    : SecretService (GNOME Keyring / KWallet)
keyring>=24.0.0,<26.0

# --- Windows extras (strongly recommended on Windows) ------------------------
# pywin32 enables two important features on Windows:
#   1. Accurate "edited by" file-owner tracking in the change log
#   2. VSS (Volume Shadow Copy) integration for backing up locked/open files
#      (Outlook .pst, browser databases, running app databases, etc.)
# The sys_platform marker means pip skips this automatically on Linux/macOS —
# no need to comment/uncomment anything.
pywin32>=306; sys_platform == "win32"

# --- rclone destination support (system dependency, NOT a pip package) -------
# If you use the "rclone" destination type, you must install rclone separately:
#   Windows : https://rclone.org/downloads/ (or: winget install Rclone.Rclone)
#   macOS   : brew install rclone
#   Linux   : sudo apt install rclone  /  sudo dnf install rclone
# After installing, configure a named remote with: rclone config
# BackupSys calls rclone as a subprocess — no Python package is needed.
# Without rclone on PATH, selecting the "rclone" destination will fail with
# a "rclone not found" error at backup time.
